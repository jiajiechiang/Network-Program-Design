Demo 1 Selenium 基礎：對付 JavaScript 動態渲染網頁

很多網頁（如 SPA 單頁應用）在 requests.get() 抓下來時只有空殼。我們必須使用 Selenium 控制真實的瀏覽器打開網頁，等 JS 執行完畢後再抓取資料。

demo1_selenium_basic.py
from selenium import webdriver
from selenium.webdriver.common.by import By
import time

# 啟動 Chrome 瀏覽器 (背景執行可加入 headless 參數)
options = webdriver.ChromeOptions()
# options.add_argument('--headless') # 若不想看到瀏覽器視窗彈出，可取消註解此行
driver = webdriver.Chrome(options=options)

try:
    print("正在打開動態網頁...")
    driver.get("https://example.com/dynamic-content")
    
    # 強制等待 3 秒，讓 JavaScript 有時間載入資料
    time.sleep(3)
    
    # 透過 By.CSS_SELECTOR 尋找動態生成的元素
    titles = driver.find_elements(By.CSS_SELECTOR, "h2.dynamic-title")
    
    print(f"成功抓取 {len(titles)} 筆動態資料：")
    for title in titles:
        print("-", title.text)
        
finally:
    # 確保最後一定會關閉瀏覽器，釋放記憶體
    driver.quit()

Demo 2 Selenium 進階：顯式等待與模擬點擊、滾動

死板的 time.sleep() 效率極差。進階爬蟲會使用「顯式等待 (Explicit Wait)」，只要元素一出現就立刻進行下一步。並結合模擬滑鼠滾動來載入「無限下拉」的網頁。

demo2_selenium_interaction.py
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

driver = webdriver.Chrome()
driver.get("https://example.com/infinite-scroll")

try:
    # 1. 顯式等待：最多等 10 秒，直到 id 為 'load-more-btn' 的按鈕可以被點擊
    wait = WebDriverWait(driver, 10)
    btn = wait.until(EC.element_to_be_clickable((By.ID, "load-more-btn")))
    
    # 模擬點擊按鈕
    btn.click()
    
    # 2. 模擬滑鼠滾動到頁面最底端 (觸發 Ajax 載入更多資料)
    driver.execute_script("window.scrollTo(0, document.body.scrollHeight);")
    
    # 等待新資料出現 (假設新資料的 class 為 'new-item')
    wait.until(EC.presence_of_element_located((By.CLASS_NAME, "new-item")))
    
    print("成功載入並擷取隱藏資料！")

finally:
    driver.quit()

Demo 3 極速爬蟲：非同步 (Asyncio + Aiohttp)

如果你要抓取 1000 個網頁，傳統的 requests 是一個一個排隊抓，非常慢。使用 非同步 (Asynchronous) 技術，可以在等待伺服器回應時，同時發出其他網頁的請求，速度可提升十倍以上！

demo3_asyncio_aiohttp.py
import asyncio
import aiohttp
import time

# 定義一個非同步的抓取函式
async def fetch_page(session, url):
    async with session.get(url) as response:
        # 等待讀取回應內容
        return await response.text()

async def main():
    urls = [f"https://example.com/page/{i}" for i in range(1, 51)] # 產生 50 個網址
    
    async with aiohttp.ClientSession() as session:
        # 建立 50 個非同步任務，同時發出請求
        tasks = [fetch_page(session, url) for url in urls]
        
        # asyncio.gather 會等待所有任務完成，並收集結果
        results = await asyncio.gather(*tasks)
        
        print(f"成功抓取了 {len(results)} 個網頁！")

if __name__ == "__main__":
    start_time = time.time()
    # 執行非同步主程式
    asyncio.run(main())
    print(f"總共耗時：{time.time() - start_time:.2f} 秒")

Demo 4 工業級爬蟲框架：Scrapy 初體驗

Scrapy 是一個強大且完整的爬蟲框架。它內建了非同步處理、資料清洗 (Items)、輸出 (Pipeline) 等功能，適合開發大型且需長期維護的爬蟲專案。

demo4_scrapy_spider.py
import scrapy

# 在終端機執行: scrapy runspider demo4_scrapy_spider.py -o output.json

class MySpider(scrapy.Spider):
    # 爬蟲的唯一名稱
    name = 'my_spider'
    # 起始網址
    start_urls = ['https://quotes.toscrape.com/']

    # 下載完成後會自動呼叫 parse 進行解析
    def parse(self, response):
        # 使用 CSS 選擇器提取資料
        for quote in response.css('div.quote'):
            yield {
                'text': quote.css('span.text::text').get(),
                'author': quote.css('span small::text').get(),
            }
            
        # 自動尋找下一頁的連結並繼續爬取
        next_page = response.css('li.next a::attr(href)').get()
        if next_page is not None:
            yield response.follow(next_page, self.parse)

Demo 5 突破反爬蟲：隨機 User-Agent 與 Proxy 代理池

為避免被伺服器封鎖 IP，進階爬蟲必須實作「代理伺服器 (Proxy)」輪替，並使用 `fake-useragent` 套件每次產生不同的瀏覽器指紋。

demo5_anti_scraping.py
import requests
from fake_useragent import UserAgent
import random

ua = UserAgent()
url = "https://httpbin.org/ip" # 用來測試當前 IP 的網站

# 準備你的 Proxy 代理池 (這裡為範例，實務上需購買或尋找可用 Proxy)
proxies_list = [
    "http://123.45.67.89:8080",
    "http://98.76.54.32:3128"
]

for i in range(3):
    # 1. 隨機產生一個 User-Agent
    headers = {"User-Agent": ua.random}
    
    # 2. 隨機挑選一個 Proxy
    proxy = random.choice(proxies_list)
    proxies = {
        "http": proxy,
        "https": proxy
    }
    
    try:
        print(f"使用 Proxy: {proxy}, UA: {headers['User-Agent'][:20]}...")
        # 帶入 headers 與 proxies
        response = requests.get(url, headers=headers, proxies=proxies, timeout=5)
        print("伺服器看到的你的 IP：", response.json()['origin'])
    except Exception as e:
        print("Proxy 連線失敗，請更換節點。")

Demo 6 現代爬蟲救星：Playwright 攔截網路請求 (API)

Playwright 是比 Selenium 更新、更快的新一代自動化工具。更厲害的是，它可以直接監聽並「攔截」網頁背景發出的 API 請求，讓你連 HTML 都不用解析，直接從封包中劫走 JSON 數據！

demo6_playwright_intercept.py
from playwright.sync_api import sync_playwright

def run(playwright):
    browser = playwright.chromium.launch(headless=True)
    page = browser.new_page()

    # 定義事件處理函式：當網頁發出任何 Response 時觸發
    def handle_response(response):
        # 判斷網址是否是我們要找的背景 API
        if "/api/v1/hidden-data" in response.url and response.status == 200:
            print("【成功劫截 API 數據】")
            print(response.json()) # 直接拿到最乾淨的 JSON 數據

    # 綁定監聽事件
    page.on("response", handle_response)

    print("打開網頁，等待網頁自動載入數據...")
    page.goto("https://example.com/complex-dashboard")
    
    # 等待網頁完全載入
    page.wait_for_load_state("networkidle")
    
    browser.close()

with sync_playwright() as playwright:
    run(playwright)
