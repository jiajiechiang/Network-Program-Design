Demo 1 企業級 NETCONF 協定：透過 ncclient 取得設備 YANG 模型狀態

NETCONF (Network Configuration Protocol) 是運行在 SSH 之上的 XML 協定，透過 YANG 數據模型提供強大且結構化的網路設備管理能力。

netconf_get_capabilities.py
from ncclient import manager
import sys

device_params = {
    'host': '192.168.1.1',
    'port': 830,  # NETCONF 預設 Port
    'username': 'admin',
    'password': 'secret',
    'hostkey_verify': False
}

try:
    print("正在透過 NETCONF 連線至設備...")
    with manager.connect(**device_params) as m:
        print("✅ NETCONF 連線成功！支援的 YANG 模型能力 (Capabilities)：")
        for capability in m.server_capabilities:
            if 'yang' in capability.lower():
                print(f"  - {capability}")
                
except Exception as e:
    print(f"NETCONF 連線失敗：{e}")

Demo 2 NETCONF 交易式組態派發：具備 Commit 與 Rollback 機制的設定

NAPALM 的 Rollback 是事後還原，而 NETCONF 支援內建的 ACID 交易保證。您可以將設定推送到 candidate 資料庫，測試無誤後才 Commit，若連線中斷會自動回滾。

netconf_transaction_config.py
from ncclient import manager

device_params = {
    'host': '192.168.1.1',
    'port': 830,
    'username': 'admin',
    'password': 'secret',
    'hostkey_verify': False
}

# 準備標準的 XML 設定檔內容 (以 YANG 介面為例)
config_xml = """

  
    
      Loopback100
      ianaift:softwareLoopback
      true
    
  

"""

with manager.connect(**device_params) as m:
    try:
        # 鎖定 running 資料庫，防止其他人同時修改
        m.lock(target='candidate')
        
        # 將設定推送至 candidate
        m.edit_config(target='candidate', config=config_xml)
        print("設定已推送至候選區 (Candidate)，正在進行交易提交...")
        
        # 提交設定 (Commit)
        m.commit()
        print("✅ NETCONF 交易提交成功！")
        
        m.unlock(target='candidate')
    except Exception as e:
        print(f"❌ 發生錯誤，自動執行 Rollback 回滾：{e}")
        m.discard_changes()
        m.unlock(target='candidate')

Demo 4 OAuth 2.0 自動續期機制：管理具備時效性的雲端網管 API

當串接現代雲端網管平台 (如 Cisco Meraki Dashboard API 或 AWS VPC API) 時，Token 通常在 1 小時內就會過期。此範例教您寫一個具備「自動偵測過期並重新換取 Token」的智能 API 請求封裝類別。

oauth_auto_refresh_client.py
import requests
import time

class SmartCloudAPIClient:
    def __init__(self, token_url, client_id, client_secret):
        self.token_url = token_url
        self.client_id = client_id
        self.client_secret = client_secret
        self.access_token = None
        self.token_expiry = 0
        
    def _refresh_token(self):
        print("🔄 偵測到 OAuth 2.0 Token 已過期或尚未取得，正在重新換取...")
        response = requests.post(self.token_url, data={
            'grant_type': 'client_credentials',
            'client_id': self.client_id,
            'client_secret': self.client_secret
        })
        if response.status_code == 200:
            data = response.json()
            self.access_token = data.get('access_token')
            # 設定過期時間 (假設有效期間 3600 秒，提前 60 秒刷新)
            self.token_expiry = time.time() + data.get('expires_in', 3600) - 60
            print("✅ 成功刷新 OAuth Token！")
        else:
            raise Exception(f"Token 換取失敗: {response.text}")
            
    def get_resource(self, api_endpoint):
        # 若 Token 過期或不存在，自動觸發刷新
        if not self.access_token or time.time() >= self.token_expiry:
            self._refresh_token()
            
        headers = {'Authorization': f'Bearer {self.access_token}'}
        return requests.get(api_endpoint, headers=headers)

Demo 5 網管微服務架構：編寫自動化 API 網關 (API Gateway)

在大型企業中，我們常需要建立一個統一的 Python API 網關，將不同廠牌設備的底層 API 封裝成統一格式，供內部前端或維運系統呼叫。

network_api_gateway.py
from flask import Flask, request, jsonify
import requests

app = Flask(__name__)

# 企業內部統一網管 API 網關
@app.route('/gateway/device/vlan', methods=['POST'])
def gateway_create_vlan():
    req_data = request.json
    target_router_ip = req_data.get('ip')
    vlan_id = req_data.get('vlan_id')
    vlan_name = req_data.get('vlan_name')
    
    print(f"🌐 [API Gateway] 收到來自前端的統一請求：在 {target_router_ip} 建立 VLAN {vlan_id}")
    
    # 網關在背景將其轉譯為設備看得懂的 RESTCONF 請求
    device_restconf_url = f"https://{target_router_ip}/restconf/data/Cisco-IOS-XE-native:native/vlan"
    payload = {
        "Cisco-IOS-XE-native:vlan-list": [{"id": vlan_id, "name": vlan_name}]
    }
    
    try:
        res = requests.post(
            device_restconf_url, 
            json=payload, 
            headers={"Content-Type": "application/yang-data+json"},
            auth=('admin', 'secret'), 
            verify=False, 
            timeout=5
        )
        return jsonify({
            "gateway_status": "success",
            "device_response_code": res.status_code,
            "message": f"VLAN {vlan_id} 已透過網關成功派發！"
        }), 200
    except Exception as e:
        return jsonify({"gateway_status": "error", "message": str(e)}), 500

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=8000)

Demo 6 分散式 API 任務隊列：結合 Celery 與 REST API 實現非同步異動

當透過 API 異動全網 500 台交換器設定時，若在 HTTP 請求中同步等待會導致連線逾時。高階架構必須結合 Celery + Redis，將 API 異動請求轉為背景非同步任務處理。

celery_api_tasks.py
from celery import Celery
import requests
import urllib3
urllib3.disable_warnings(urllib3.exceptions.InsecureRequestWarning)

celery_app = Celery(
    'net_api_tasks',
    broker='redis://localhost:6379/0',
    backend='redis://localhost:6379/0'
)

@celery_app.task
def async_push_restconf_config(device_ip, payload):
    print(f"🚀 [Celery Worker] 開始非同步推送設定至設備 {device_ip}...")
    url = f"https://{device_ip}/restconf/data/ietf-interfaces:interfaces"
    
    try:
        response = requests.patch(
            url, 
            json=payload, 
            headers={"Content-Type": "application/yang-data+json"},
            auth=('admin', 'secret'), 
            verify=False,
            timeout=10
        )
        print(f"✅ 設備 {device_ip} 非同步 API 推送完成，狀態碼：{response.status_code}")
        return response.status_code
    except Exception as e:
        print(f"❌ 設備 {device_ip} 非同步推送異常：{e}")
        return str(e)
