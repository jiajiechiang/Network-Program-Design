Demo 1 打造即時 Syslog 伺服器：攔截 SSH 暴力破解攻擊 (Brute-force)

駭客常對設備發動 SSH 密碼猜解。我們可以使用 Python 的 Socket 撰寫一個輕量級 Syslog 監聽伺服器，當收到登入失敗日誌達一定次數時，自動觸發防禦機制。

syslog_security_monitor.py
import socket
import re

HOST = '0.0.0.0'  # 監聽所有網卡
PORT = 514        # 標準 Syslog UDP Port (需管理員權限或改用 5140)

failed_attempts = {}

with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as server:
    server.bind((HOST, PORT))
    print(f"🚨 網管即時 Syslog 監聽伺服器已啟動於 Port {PORT}...")
    
    while True:
        data, addr = server.recvfrom(1024)
        log_message = data.decode('utf-8', errors='ignore')
        
        # 檢查是否為 Cisco 設備的 SSH 登入失敗日誌 (例如 %AUTH-4-FAIL 或 %SSH-5-FAILED)
        if 'failed' in log_message.lower() or 'invalid' in log_message.lower():
            attacker_ip = addr[0]
            print(f"⚠️ [收到可疑日誌] 來源 IP: {attacker_ip} | 內容: {log_message.strip()}")
            
            # 計算攻擊次數
            failed_attempts[attacker_ip] = failed_attempts.get(attacker_ip, 0) + 1
            
            if failed_attempts[attacker_ip] >= 5:
                print(f"🚨 [安全警報] 偵測來自 {attacker_ip} 的持續暴力破解攻擊！準備發動防禦...")

Demo 2 控制平面防禦 (CoPP) 稽核：防止路由器 CPU 遭 DoS 癱瘓

核心路由器若未設定 CoPP (Control Plane Policing)，當遭到大量 Ping 或異常流量攻擊時，CPU 會飆高到 100% 導致整網斷線。此範例自動檢查設備是否套用 CoPP 防禦策略。

audit_copp_policy.py
from netmiko import ConnectHandler

device = {
    'device_type': 'cisco_ios',
    'host': '192.168.1.1',
    'username': 'admin',
    'password': 'secret'
}

with ConnectHandler(**device) as conn:
    output = conn.send_command('show running-config | include control-plane')
    policy_output = conn.send_command('show policy-map control-plane')
    
    print("=== 控制平面防禦 (CoPP) 稽核報告 ===")
    if 'control-plane' in output and policy_output.strip():
        print("✅ [安全] 設備已配置 Control Plane Policing (CoPP) 保護路由核心。")
        print(policy_output[:300])
    else:
        print("🚨 [嚴重資安隱患] 設備未啟用 CoPP！核心 CPU 暴露於 DoS 攻擊風險中。")

Demo 3 SSL/TLS 憑證有效期限監控：防止設備管理介面憑證過期

網管人員常因 HTTPS 管理介面的 SSL 憑證過期而無法安全登入。此範例透過 Python 自動連線至設備的 HTTPS 埠，檢查憑證剩餘有效天數，過期前自動發出警報。

monitor_ssl_certificate.py
import socket
import ssl
import datetime

HOST = '192.168.1.1'
PORT = 443

context = ssl.create_default_context()
context.check_hostname = False
context.verify_mode = ssl.CERT_NONE # 略過自簽憑證驗證

try:
    print(f"正在檢查 {HOST}:{PORT} 的 SSL 憑證效期...")
    with socket.create_connection((HOST, PORT), timeout=5) as sock:
        with context.wrap_socket(sock, server_hostname=HOST) as ssock:
            cert = ssock.getpeercert(binary_form=True)
            
            # 使用 cryptography 解析憑證過期時間
            from cryptography import x509
            from cryptography.hazmat.backends import default_backend
            
            x509_cert = x509.load_der_x509_certificate(cert, default_backend())
            expire_date = x509_cert.not_after_utc if hasattr(x509_cert, 'not_after_utc') else x509_cert.not_after
            
            remaining_days = (expire_date.replace(tzinfo=None) - datetime.datetime.utcnow()).days
            
            print(f"📅 憑證有效期限至：{expire_date}")
            print(f"⏳ 剩餘有效天數：{remaining_days} 天")
            
            if remaining_days < 30:
                print("⚠️ [憑證警告] 剩餘有效天數不足 30 天，請儘速更新憑證！")
            else:
                print("✅ 憑證效期正常。")
                
except Exception as e:
    print(f"無法取得 SSL 憑證：{e}")

Demo 4 路由表劫持防禦：自動監控 BGP 路由異動與異常路徑

BGP 路由劫持 (BGP Hijacking) 是嚴重的網路資安事件。此範例透過 NAPALM 實時抓取 BGP 鄰居狀態與路由表，比對預期的白名單路徑，發現異常立刻通報。

monitor_bgp_hijack.py
from napalm import get_network_driver

driver = get_network_driver('ios')
device = driver(hostname='192.168.1.1', username='admin', password='secret')
device.open()

# 取得當前 BGP 鄰居狀態
bgp_neighbors = device.get_bgp_neighbors()

print("=== BGP 鄰居安全監控 ===")
for neighbor_ip, details in bgp_neighbors['global']['peers'].items():
    is_up = details.get('is_up')
    uptime = details.get('uptime')
    
    if is_up:
        print(f"✅ BGP 鄰居 {neighbor_ip} 狀態正常 (Uptime: {uptime})")
    else:
        print(f"🚨 [嚴重警報] BGP 鄰居 {neighbor_ip} 已經斷線！可能遭受阻斷攻擊或路由表被竄改！")

device.close()

Demo 5 跨網段多執行緒資安漏洞掃描 (Nornir + Netmiko)

企業有數百台設備需要定期執行資安合規檢查。結合 Nornir 的多執行緒併發能力，瞬間對全網設備發動資安巡檢，幾秒鐘產出全公司資安紅黑榜。

nornir_security_scanner.py
from nornir import InitNornir
from nornir_netmiko.tasks import netmiko_send_command
from nornir_utils.plugins.functions import print_result

nr = InitNornir(config_file="config.yaml")

def security_compliance_audit(task):
    # 檢查是否開啟 password-encryption 與 SNMP 預設字串
    res_enc = task.run(task=netmiko_send_command, command_string="show running-config | include password-encryption")
    res_snmp = task.run(task=netmiko_send_command, command_string="show running-config | include snmp-server community")
    
    task.host["audit_results"] = {
        "pwd_encryption": "service password-encryption" in res_enc.result,
        "default_snmp": "public" in res_snmp.result or "private" in res_snmp.result
    }

results = nr.run(task=security_compliance_audit)

for host_name, host_obj in nr.inventory.hosts.items():
    data = host_obj.get("audit_results", {})
    print(f"設備 [{host_name}] 稽核結果 -> 密碼加密: {data.get('pwd_encryption')} | 高危SNMP預設字串: {data.get('default_snmp')}")

Demo 6 資安事件自動阻斷：結合 Webhook 與防火牆 API 實時封鎖

整合企業的資安防禦網。當 Python 監控腳本偵測到惡意 IP 或異常設備嘗試連線時，自動呼叫次世代防火牆 (如 Palo Alto / Fortinet) 的 REST API 將其加入黑名單，實現零秒反應的自動化防禦 (SOAR)。

soar_auto_block.py
import requests

def block_malicious_ip_on_firewall(malicious_ip):
    # 模擬企業防火牆 (例如 Fortinet / Palo Alto) 的 REST API 接口
    firewall_api_url = "https://firewall.corporate.local/api/v2/firewall/address_block"
    
    headers = {
        "Authorization": "Bearer SECRET_API_TOKEN_XYZ",
        "Content-Type": "application/json"
    }
    
    payload = {
        "ip_address": malicious_ip,
        "action": "drop",
        "reason": "Auto-quarantined by Python SOAR Security Script"
    }
    
    try:
        # 發送 API 請求封鎖該 IP
        response = requests.post(firewall_api_url, json=payload, verify=False, timeout=5)
        
        if response.status_code == 200:
            print(f"🛡️ [SOAR 自動防禦] 惡意 IP {malicious_ip} 已被防火牆即時封鎖！")
        else:
            print(f"⚠️ 防火牆 API 回應異常：{response.status_code}")
            
    except Exception as e:
        print(f"無法連線至防火牆 API：{e}")

# 模擬攔截到攻擊來源
block_malicious_ip_on_firewall('203.0.113.55')
